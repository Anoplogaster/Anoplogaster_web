{{- $image := .image -}} {{- $resource := resources.Get (printf "media/photography/%s" $image.file) -}} {{- if not $resource -}} {{- errorf "Photography image not found: %s" $image.file -}} {{- end -}} {{- $label := $image.common | default $image.scientific | default "photograph" -}} {{- $orientation := "square" -}} {{- if gt (mul $resource.Width 100) (mul $resource.Height 120) -}} {{- $orientation = "landscape" -}} {{- else if gt (mul $resource.Height 100) (mul $resource.Width 120) -}} {{- $orientation = "portrait" -}} {{- end -}} {{/* Resize/Fit preserve the original aspect; never enlarge a source. */}} {{- $thumbnailSet := slice -}} {{- $thumbnail := false -}} {{- $lastWidth := 0 -}} {{- range slice 360 640 960 -}} {{- $target := cond (lt $resource.Width .) $resource.Width . -}} {{- $candidate := $resource.Resize (printf "%dx webp q90" $target) -}} {{- if not $thumbnail -}}{{- $thumbnail = $candidate -}}{{- end -}} {{- if gt $candidate.Width $lastWidth -}} {{- $thumbnailSet = $thumbnailSet | append (printf "%s %dw" $candidate.RelPermalink $candidate.Width) -}} {{- $lastWidth = $candidate.Width -}} {{- end -}} {{- end -}} {{- $viewerSet := slice -}} {{- $viewer := false -}} {{- $lastWidth = 0 -}} {{- $longEdge := $resource.Width -}} {{- if gt $resource.Height $longEdge -}}{{- $longEdge = $resource.Height -}}{{- end -}} {{- range slice 1200 2000 2560 -}} {{- $target := cond (lt $longEdge .) $longEdge . -}} {{- $candidate := $resource.Fit (printf "%dx%d webp q90" $target $target) -}} {{- if not $viewer -}}{{- $viewer = $candidate -}}{{- end -}} {{- if gt $candidate.Width $lastWidth -}} {{- $viewerSet = $viewerSet | append (printf "%s %dw" $candidate.RelPermalink $candidate.Width) -}} {{- $lastWidth = $candidate.Width -}} {{- end -}} {{- end -}}
