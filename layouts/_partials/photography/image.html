{{- $image := .image -}} {{- $resource := resources.Get (printf "media/photography/%s" $image.file) -}} {{- if not $resource -}} {{- errorf "Photography image not found: %s" $image.file -}} {{- end -}} {{- $smallTarget := cond (lt $resource.Width 480) $resource.Width 480 -}} {{- $largeTarget := cond (lt $resource.Width 800) $resource.Width 800 -}} {{- $small := $resource.Resize (printf "%dx webp q82" $smallTarget) -}} {{- $large := $resource.Resize (printf "%dx webp q86" $largeTarget) -}} {{- $full := $resource.Process "webp q90" -}} {{- $orientation := "square" -}} {{- if gt (mul $resource.Width 100) (mul $resource.Height 120) -}} {{- $orientation = "landscape" -}} {{- else if gt (mul $resource.Height 100) (mul $resource.Width 120) -}} {{- $orientation = "portrait" -}} {{- end -}} {{- $label := $image.common | default $image.scientific | default "photograph" -}} {{- $srcset := slice (printf "%s %dw" $small.RelPermalink $small.Width) -}} {{- if gt $large.Width $small.Width -}} {{- $srcset = $srcset | append (printf "%s %dw" $large.RelPermalink $large.Width) -}} {{- end -}} {{- if gt $full.Width $large.Width -}} {{- $srcset = $srcset | append (printf "%s %dw" $full.RelPermalink $full.Width) -}} {{- end -}}
